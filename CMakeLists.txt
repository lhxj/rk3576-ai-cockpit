cmake_minimum_required(VERSION 3.21)
project(rk3576_ai_cockpit VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(BUILD_COCKPIT_UI "Build the cockpit UI foundation and optional Qt shell" ON)
add_subdirectory(libs/common)
if(BUILD_COCKPIT_UI)
  add_subdirectory(apps/cockpit_ui)
endif()
add_executable(cockpit_host_smoke tools/host_smoke/main.cpp)
target_link_libraries(cockpit_host_smoke PRIVATE cockpit_common)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cockpit_host_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(cockpit_common_test tests/unit/common_test.cpp)
  target_link_libraries(cockpit_common_test PRIVATE cockpit_common)
  add_test(NAME common_metadata COMMAND cockpit_common_test)
  add_test(NAME scaffold_smoke COMMAND cockpit_host_smoke --self-test)
endif()
message(STATUS "Hardware Camera/ALSA/NPU/RPMsg backends are not part of this build")
