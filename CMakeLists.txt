cmake_minimum_required(VERSION 3.21)
project(rk3576_ai_cockpit VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Threads REQUIRED)
option(COCKPIT_ENABLE_SHERPA_ASR "Build the external Sherpa C API file backend" OFF)
add_subdirectory(libs/common)
add_subdirectory(libs/protocol)
add_subdirectory(libs/ipc)
add_subdirectory(apps/audio_srv)
add_subdirectory(libs/audio)
add_subdirectory(apps/voice_srv)
add_subdirectory(apps/infer_srv)
add_executable(cockpit_host_smoke tools/host_smoke/main.cpp)
target_link_libraries(cockpit_host_smoke PRIVATE cockpit_common)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cockpit_host_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(cockpit_common_test tests/unit/common_test.cpp)
  target_link_libraries(cockpit_common_test PRIVATE cockpit_common)
  add_test(NAME common_metadata COMMAND cockpit_common_test)
  add_test(NAME scaffold_smoke COMMAND cockpit_host_smoke --self-test)
  function(add_foundation_test name source library)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE ${library})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endfunction()
  add_foundation_test(protocol_test tests/unit/protocol/protocol_test.cpp cockpit_protocol)
  add_foundation_test(ipc_test tests/unit/ipc/ipc_test.cpp cockpit_ipc)
  add_foundation_test(audio_test tests/unit/audio/audio_test.cpp cockpit_audio)
  add_foundation_test(voice_test tests/unit/voice/voice_test.cpp cockpit_voice)
  add_foundation_test(infer_test tests/unit/infer/infer_test.cpp cockpit_infer)
  add_foundation_test(wav_reader_test tests/unit/audio/wav_reader_test.cpp cockpit_audio_file)
  add_foundation_test(sherpa_asr_test tests/unit/voice/sherpa_asr_test.cpp cockpit_sherpa_asr)
endif()
message(STATUS "Host Voice/AI: Sherpa file ASR ${COCKPIT_ENABLE_SHERPA_ASR}; no ALSA/TTS/RKLLM/RKNN backend")
