if(COCKPIT_ENABLE_RKNN)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "COCKPIT_ENABLE_RKNN requires Linux")
  endif()
  set(RKNN_INCLUDE_DIR "" CACHE PATH "Directory containing rknn_api.h")
  set(RKNN_LIBRARY "" CACHE FILEPATH "Path to librknnrt.so")
  if(NOT RKNN_INCLUDE_DIR)
    find_path(RKNN_INCLUDE_DIR rknn_api.h)
  endif()
  if(NOT RKNN_LIBRARY)
    find_library(RKNN_LIBRARY NAMES rknnrt)
  endif()
  if(NOT RKNN_INCLUDE_DIR OR NOT EXISTS "${RKNN_INCLUDE_DIR}/rknn_api.h")
    message(FATAL_ERROR "COCKPIT_ENABLE_RKNN requires RKNN_INCLUDE_DIR/rknn_api.h")
  endif()
  if(NOT RKNN_LIBRARY)
    message(FATAL_ERROR "COCKPIT_ENABLE_RKNN requires RKNN_LIBRARY/librknnrt.so")
  endif()
  add_library(cockpit_rknn_vision STATIC src/rknn_vision_backend.cpp)
  target_include_directories(cockpit_rknn_vision PUBLIC include PRIVATE ${RKNN_INCLUDE_DIR})
  target_link_libraries(cockpit_rknn_vision PUBLIC cockpit_vision_runtime ${RKNN_LIBRARY})
  target_compile_features(cockpit_rknn_vision PUBLIC cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cockpit_rknn_vision PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  message(STATUS "RKNN vision enabled: include=${RKNN_INCLUDE_DIR}; library=${RKNN_LIBRARY}")
endif()
