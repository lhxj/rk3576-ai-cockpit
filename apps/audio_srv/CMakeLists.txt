if(COCKPIT_ENABLE_ALSA_CAPTURE)
  find_path(ALSA_INCLUDE_DIR alsa/asoundlib.h)
  find_library(ALSA_LIBRARY NAMES asound)
  if(NOT ALSA_INCLUDE_DIR OR NOT ALSA_LIBRARY)
    message(FATAL_ERROR "ALSA capture requested but libasound development dependency not found")
  endif()
  add_library(cockpit_alsa_capture STATIC src/alsa_capture.cpp)
  target_include_directories(cockpit_alsa_capture PUBLIC include PRIVATE "${ALSA_INCLUDE_DIR}")
  target_link_libraries(cockpit_alsa_capture PUBLIC cockpit_audio "${ALSA_LIBRARY}")
  target_compile_features(cockpit_alsa_capture PUBLIC cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cockpit_alsa_capture PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
